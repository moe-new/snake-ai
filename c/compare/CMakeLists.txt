cmake_minimum_required(VERSION 3.20)
project(snake_compare LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BENCH_SANITIZE "Build validation executables with sanitizers" OFF)
add_library(snake_agents STATIC src/game.c src/bfs.c src/tiled.c src/cycle.c src/cell.c)
target_include_directories(snake_agents PUBLIC src)
add_executable(snake-compare src/main.c)
target_link_libraries(snake-compare PRIVATE snake_agents)
if(NOT BENCH_SANITIZE)
  set_target_properties(snake-compare PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(snake-compare PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}")
  endforeach()
endif()
add_executable(snake-tests src/tests.c)
target_link_libraries(snake-tests PRIVATE snake_agents)
foreach(target snake_agents snake-compare snake-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W3 /utf-8 /fp:precise)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(BENCH_SANITIZE)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-function -Wno-misleading-indentation -ffp-contract=off)
    if(BENCH_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    endif()
  endif()
endforeach()
if(BENCH_SANITIZE)
  foreach(target snake-compare snake-tests)
    if(MSVC)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    else()
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
enable_testing()
add_test(NAME invariants COMMAND snake-tests)
